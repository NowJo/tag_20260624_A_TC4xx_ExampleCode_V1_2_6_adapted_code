UCB calculations for TC4Dx.
The CRC checksum calculations were done with https://crccalc.com/


################################################################################################
UCB0_BMHD0, RTC
    .sal               = 0xAE404800, // System Address of this location
    .bmIDIndex         = 0xB359000F, // ID + internal start from flash, Bit 0 reserved (=1)
    .startAddr         = 0xA0000020, // Internal start from flash address
    .crcHead           = 0xAB562CEE, // the header CRC

AE404800 B359000F A0000020

################################################################################################

    .startAddr         = 0xA0000020, // Internal start from flash address
    .crcHead           = 0x9F2FEDF2, // the header CRC

AE405000 B359000F A0000020

    .startAddr         = 0xA4000000, // Internal start from flash address
    .crcHead           = 0xD1AF28A6, // the header CRC

AEC01800 B359000F A4000000
